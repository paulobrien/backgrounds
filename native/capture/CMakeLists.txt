cmake_minimum_required(VERSION 3.21)
project(BackgroundsCapture LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "MSVC runtime linkage" FORCE)
endif()
add_executable(BackgroundsCapture capture.cpp)
target_compile_definitions(BackgroundsCapture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_features(BackgroundsCapture PRIVATE cxx_std_17)
target_link_libraries(BackgroundsCapture PRIVATE mfplat mfreadwrite mf mfuuid ole32)
